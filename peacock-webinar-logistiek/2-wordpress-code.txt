<script>
(function () {
  "use strict";
  /* ================= INSTELLINGEN ================= */
  /* URL van je Power Automate-flow ("When a HTTP request is received") of een andere webhook.
     Leeg laten = de aanmelding opent een vooraf ingevulde e-mail naar FALLBACK_EMAIL. Zie LEESMIJ.md. */
  var ENDPOINT = "";
  var FALLBACK_EMAIL = "info@peacock-ia.nl";
  /* ================================================ */
  var root = document.getElementById("pck-webinar");
  var form = document.getElementById("pck-form");
  var card = document.getElementById("pck-formcard");
  if (!root || !form || !card) { return; }
  var webinar = root.getAttribute("data-webinar") || "Webinar";
  var datum = root.getAttribute("data-datum") || "";
  var tijd = root.getAttribute("data-tijd") || "";
  if (datum.indexOf("DATUM") !== -1 || tijd.indexOf("TIJD") !== -1) {
    console.warn("Peacock webinar: vergeet [DATUM] en [TIJD] niet in te vullen.");
  }
  var button = form.querySelector("button");
  var status = form.querySelector(".pck-status");
  var emailPattern = /^\S+@\S+\.\S{2,}$/;
  form.noValidate = true;
  function field(name) { return form.elements.namedItem(name); }
  function value(name) { var el = field(name); return el ? String(el.value || "").trim() : ""; }
  function setError(el, isError) {
    var wrap = el.closest(".pck-field") || el.closest(".pck-consent-wrap");
    if (wrap) { wrap.classList.toggle("is-error", isError); }
    var consent = el.closest(".pck-consent");
    if (consent) { consent.classList.toggle("is-error", isError); }
    el.setAttribute("aria-invalid", isError ? "true" : "false");
  }
  function validate() {
    var firstInvalid = null;
    var rules = [
      { name: "voornaam", ok: value("voornaam") !== "" },
      { name: "achternaam", ok: value("achternaam") !== "" },
      { name: "email", ok: emailPattern.test(value("email")) },
      { name: "bedrijf", ok: value("bedrijf") !== "" },
      { name: "akkoord", ok: field("akkoord").checked }
    ];
    rules.forEach(function (rule) {
      var el = field(rule.name);
      setError(el, !rule.ok);
      if (!rule.ok && !firstInvalid) { firstInvalid = el; }
    });
    if (firstInvalid) { firstInvalid.focus(); }
    return !firstInvalid;
  }
  function collect() {
    var docs = [];
    form.querySelectorAll("input.pck-doc:checked").forEach(function (cb) { docs.push(cb.value); });
    var qs = new URLSearchParams(window.location.search);
    return {
      webinar: webinar,
      datum: datum,
      tijd: tijd,
      voornaam: value("voornaam"),
      achternaam: value("achternaam"),
      email: value("email"),
      bedrijf: value("bedrijf"),
      functie: value("functie"),
      telefoon: value("telefoon"),
      documenten: docs.join(", "),
      akkoord_privacy: true,
      pagina: window.location.origin + window.location.pathname,
      utm_source: qs.get("utm_source") || "",
      utm_medium: qs.get("utm_medium") || "",
      utm_campaign: qs.get("utm_campaign") || "",
      aangemeld_op: new Date().toISOString()
    };
  }
  function showStatus(message) {
    if (!status) { if (message) { window.alert(message); } return; }
    status.textContent = message;
    status.classList.toggle("is-visible", !!message);
  }
  function setBusy(isBusy) {
    button.classList.toggle("is-busy", isBusy);
    button.disabled = isBusy;
    button.textContent = isBusy ? "Bezig met aanmelden..." : "Ja, ik meld me aan";
  }
  function track(data) {
    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push({ event: "webinar_aanmelding", webinar: data.webinar, bedrijf: data.bedrijf });
  }
  function showSuccess(data, viaMail) {
    var title = card.querySelector(".pck-success__title");
    var text = card.querySelector(".pck-success__text");
    if (viaMail) {
      title.textContent = "Nog één stap!";
      text.textContent = "Je e-mailprogramma is geopend met je aanmelding. Verstuur die e-mail om je plek te bevestigen. Lukt dat niet? Mail je aanmelding naar " + FALLBACK_EMAIL + ".";
    } else if (data) {
      title.textContent = "Je bent aangemeld!";
      text.textContent = "Bedankt, " + data.voornaam + ". We sturen de deelnamelink naar " + data.email + ". Tot dan!";
    }
    card.classList.add("is-done");
    card.scrollIntoView({ behavior: "smooth", block: "start" });
    title.focus({ preventScroll: true });
  }
  function sendByMail(data) {
    var lines = [
      "Graag meld ik me aan voor: " + data.webinar + " (" + data.datum + ", " + data.tijd + ").",
      "",
      "Naam: " + data.voornaam + " " + data.achternaam,
      "E-mail: " + data.email,
      "Bedrijf: " + data.bedrijf,
      "Functie: " + data.functie,
      "Telefoon: " + data.telefoon,
      "Documenten die de meeste tijd kosten: " + data.documenten,
      "",
      "Ik ga akkoord met de verwerking van mijn gegevens volgens de privacyverklaring van Peacock."
    ];
    window.location.href = "mailto:" + FALLBACK_EMAIL +
      "?subject=" + encodeURIComponent("Aanmelding " + data.webinar) +
      "&body=" + encodeURIComponent(lines.join("\n"));
    track(data);
    showSuccess(data, true);
  }
  form.addEventListener("submit", function (event) {
    event.preventDefault();
    showStatus("");
    if (value("website") !== "") { showSuccess(null, false); return; }
    if (!validate()) { return; }
    var data = collect();
    if (!ENDPOINT) { sendByMail(data); return; }
    setBusy(true);
    fetch(ENDPOINT, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(data)
    }).then(function (response) {
      if (!response.ok) { throw new Error("HTTP " + response.status); }
      track(data);
      showSuccess(data, false);
    }).catch(function () {
      setBusy(false);
      showStatus("Er ging iets mis bij het versturen. Probeer het opnieuw of mail ons via " + FALLBACK_EMAIL + ".");
    });
  });
  form.addEventListener("input", function (event) {
    if (event.target.getAttribute("aria-invalid") === "true") { setError(event.target, false); }
  });
  form.addEventListener("change", function (event) {
    if (event.target.getAttribute("aria-invalid") === "true") { setError(event.target, false); }
  });
})();
</script>
